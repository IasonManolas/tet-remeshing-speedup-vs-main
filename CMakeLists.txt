cmake_minimum_required(VERSION 3.12...3.31)
project(remesh_bench CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
find_package(CGAL REQUIRED)
add_executable(remesh_bench remesh_bench.cpp)
target_link_libraries(remesh_bench PRIVATE CGAL::CGAL)
target_compile_features(remesh_bench PRIVATE cxx_std_17)
